   958	    def _classify_layers(self, map_data: dict) -> None:
   959	        """图层分类综合（QGIS/ArcGIS 图层管理）：
   960	
   961	        给每个图层补齐持久化的分类元数据：
   962	          - group   按制图要素类别分组的名称（底图/行政区划/水系/湖泊/…）
   963	          - format  颜色/线宽/填充/虚线等视觉格式摘要
   964	          - visible 图层可见性（默认True，可隐藏并持久化）
   965	
   966	        分类依据"颜色与格式 + 名称语义"，与前端图层面板联动。
   967	        """
   968	        if not map_data or "layers" not in map_data:
   969	            return
   970	        changed = False
   971	        # 老图迁移：道路图层英文名 -> 中文名 + 子分组
   972	        ROAD_CN = {
   973	            "motorway": "高速公路主线", "motorway_link": "高速互通匝道",
   974	            "trunk": "城市干线主干道", "trunk_link": "主干道连接匝道",
   975	            "primary": "城市主干道", "primary_link": "主干道衔接匝道",
   976	            "secondary": "城市次干道", "secondary_link": "次干道连接匝道",
   977	            "tertiary": "三级道路（次要道路）", "tertiary_link": "三级道路连接线",
   978	            "residential": "居民区街区道路", "living_street": "生活性街道",
   979	            "service": "服务性道路", "unclassified": "未分级道路", "other": "其他道路",
   980	        }
   981	        ROAD_SUB = {
   982	            "motorway": "高速路网", "motorway_link": "高速路网",
   983	            "trunk": "城市主干道", "trunk_link": "城市主干道",
   984	            "primary": "城市主干道", "primary_link": "城市主干道",
   985	            "secondary": "城区道路", "secondary_link": "城区道路",
   986	            "tertiary": "城区道路", "tertiary_link": "城区道路",
   987	            "residential": "城区道路", "living_street": "城区道路",
   988	            "service": "其他道路", "unclassified": "其他道路", "other": "其他道路",
   989	        }
   990	        ROAD_OPACITY = {
   991	            # 与 carto-agent-1 行政区划图完全一致：道路统一不透明度 0.9
   992	            "motorway": 0.9, "motorway_link": 0.9, "trunk": 0.9, "trunk_link": 0.9,
   993	            "primary": 0.9, "primary_link": 0.9, "secondary": 0.9,
   994	            "secondary_link": 0.9, "tertiary": 0.9, "tertiary_link": 0.9,
   995	            "residential": 0.9, "living_street": 0.9, "service": 0.9,
   996	            "unclassified": 0.9, "other": 0.9,
   997	        }
   998	
   999	        def _cat(layer: dict) -> str:
  1000	            n = layer.get("name", "") or ""
  1001	            t = layer.get("type", "") or ""
  1002	            if re.search(r"陆地底图|省域|周边地市|市域底图|湖北省", n):
  1003	                return "底图"
  1004	            if re.search(r"政区|区县|边界|行政中心|区划", n):
  1005	                return "行政区划"
  1006	            if re.search(r"河流|水系|河道|中心线|大江|水面", n):
  1007	                return "水系"
  1008	            if "湖泊" in n:
  1009	                return "湖泊"
  1010	            if "居民地" in n:
  1011	                return "居民地"
  1012	            if "等高线" in n:
  1013	                return "等高线"
  1014	            if n.startswith("道路-") or re.search(r"高速|国道|省道|主干道|次干道|支路|街巷", n):
  1015	                return "道路"
  1016	            if re.search(r"轨道|铁路|地铁|轻轨", n):
  1017	                return "轨道/铁路"
  1018	            if t in ("circleMarker", "marker", "point", "circle"):
  1019	                return "POI/符号"
  1020	            if t in ("textLabel", "label") or "注记" in n or "标注" in n:
  1021	                return "注记/标注"
  1022	            return "其他"
  1023	
  1024	        for layer in map_data["layers"]:
  1025	            # 老图道路图层命名迁移
  1026	            _m = re.match(r"^道路-([a-z_]+)$", layer.get("name", "") or "")
  1027	            if _m and _m.group(1) in ROAD_CN:
  1028	                hw = _m.group(1)
  1029	                new_name = f"道路-{ROAD_CN[hw]}"
  1030	                if layer.get("name") != new_name:
  1031	                    layer["name"] = new_name
  1032	                    changed = True
  1033	                if layer.get("subgroup") != ROAD_SUB[hw]:
  1034	                    layer["subgroup"] = ROAD_SUB[hw]
  1035	                    changed = True
  1036	                _st = layer.setdefault("style", {})
  1037	                if _st.get("opacity", 1.0) != ROAD_OPACITY.get(hw):
  1038	                    _st["opacity"] = ROAD_OPACITY.get(hw, 0.9)
  1039	                    changed = True
  1040	                layer.setdefault("metadata", {})["raw_class"] = hw
  1041	                layer.setdefault("metadata", {})["description"] = ROAD_CN[hw]
  1042	            # 道路样式迁移（按 raw_class / properties.subtype 设置分级不透明度）
  1043	            _rc = (layer.get("metadata") or {}).get("raw_class")
  1044	            if not _rc and layer.get("properties") and isinstance(layer.get("properties"), list):
  1045	                _p0 = layer["properties"][0] if layer["properties"] else None
  1046	                _rc = _p0.get("subtype") if isinstance(_p0, dict) else None
  1047	            if _rc in ROAD_OPACITY:
  1048	                _st = layer.setdefault("style", {})
  1049	                if _st.get("opacity", 1.0) != ROAD_OPACITY[_rc]:
  1050	                    _st["opacity"] = ROAD_OPACITY[_rc]
  1051	                    changed = True
  1052	            if "group" not in layer:
  1053	                layer["group"] = _cat(layer)
  1054	                changed = True
  1055	            if "format" not in layer:
  1056	                st = layer.get("style") or {}
  1057	                layer["format"] = {
  1058	                    "color": st.get("color", ""),
  1059	                    "fillColor": st.get("fillColor", ""),
  1060	                    "weight": st.get("weight"),
  1061	                    "dashArray": st.get("dashArray", ""),
  1062	                    "opacity": st.get("opacity"),
  1063	                }
  1064	                changed = True
  1065	            if "visible" not in layer:
  1066	                layer["visible"] = True
  1067	                changed = True
  1068	        if changed:
  1069	            self._schedule_save()

  1126	    def fetch_data_multi_source(
  1127	        self,
  1128	        data_type: str,
  1129	        bbox: Optional[List[float]] = None,
  1130	        region: Optional[str] = None,
  1131	        limit: int = 1000,
  1132	        preferred_source: Optional[str] = None,
  1133	        merge: bool = True,
  1134	        **filters,
  1135	    ) -> DataResult:
  1136	        """多源数据融合获取
  1137	
  1138	        通过 DataSourceRegistry 从多个数据源获取地理要素数据，
  1139	        支持自动选择最优源、多源查询、结果合并去重。
  1140	
  1141	        Args:
  1142	            data_type: 要素类型，如 "highway", "poi/restaurant", "vec"
  1143	            bbox: 边界框 [min_lng, min_lat, max_lng, max_lat]，
  1144	                  为 None 时自动根据 region 填充
  1145	            region: 区域名称（如"武汉市"），在 bbox 为 None 时使用
  1146	            limit: 返回数量上限
  1147	            preferred_source: 优先使用的数据源
  1148	            merge: 是否合并多源结果（True=合并去重, False=返回第一个源的结果）
  1149	            **filters: 额外筛选条件
  1150	
  1151	        Returns:
  1152	            DataResult: 统一返回对象
  1153	
  1154	        Example:
  1155	            # 多源融合获取武汉餐饮 POI
  1156	            result = map_service.fetch_data_multi_source(
  1157	                "poi/restaurant", region="武汉市", merge=True
  1158	            )
  1159	            # 获取 OSM 道路数据
  1160	            result = map_service.fetch_data_multi_source(
  1161	                "highway", region="武汉市", preferred_source="osm"
  1162	            )
  1163	        """
  1164	        # 构建查询对象
  1165	        if bbox:
  1166	            query = DataQuery(
  1167	                data_type=data_type,
  1168	                bbox=bbox,
  1169	                limit=limit,
  1170	                filters=filters if filters else None,
  1171	                region=region,
  1172	            )
  1173	        elif region:
  1174	            query = DataQuery.from_region(
  1175	                data_type=data_type,
  1176	                region=region,
  1177	                limit=limit,
  1178	                filters=filters if filters else None,
  1179	            )
  1180	        else:
  1181	            return DataResult.empty("map_service", "no bbox or region specified")
  1182	
  1183	        logger.info(f"[MapService] 多源数据获取: type={data_type},"
  1184	              f" region={region}, merge={merge}, preferred={preferred_source}")
  1185	
  1186	        if merge:
  1187	            # 多源融合：获取所有源的结果并合并去重
  1188	            result = self.data_registry.fetch_and_merge(query)
  1189	        elif preferred_source:
  1190	            # 指定优先源
  1191	            result = self.data_registry.fetch(query, preferred_source=preferred_source)
  1192	        else:
  1193	            # 自动选择第一个支持的源
  1194	            result = self.data_registry.fetch(query)
  1195	
  1196	        logger.info(f"[MapService] 多源数据获取完成: {result.count} 个要素,"
  1197	              f" source={result.source}, quality={result.quality_score}")
  1198	        return result

  2204	    def _connect_polylines_by_name(self, map_layers: List[dict]) -> List[dict]:
  2205	        """线状要素按名称连通：把同一条道路/边界/铁路的离散线段合并为完整折线。
  2206	
  2207	        制图综合要求不同比例尺下主要道路保持连通，不出现“一段一段”的割裂。
  2208	        方法：按要素名称分组 → 坐标吸附到 1m 网格 → unary_union + linemerge。
  2209	        """
  2210	        from shapely.geometry import LineString
  2211	        from shapely.ops import linemerge, unary_union
  2212	
  2213	        for layer in map_layers:
  2214	            if layer.get("type") not in ("polyline", "line"):
  2215	                continue
  2216	            coords = layer.get("coordinates") or []
  2217	            props = layer.get("properties") or []
  2218	            if len(coords) < 2:
  2219	                continue
  2220	            groups: Dict[str, List[Any]] = {}
  2221	            for i, c in enumerate(coords):
  2222	                if not isinstance(c, list) or len(c) < 2:
  2223	                    continue
  2224	                name = (props[i] if i < len(props) else {}).get("name") or ""
  2225	                groups.setdefault(name, []).append((c, props[i] if i < len(props) else {}))
  2226	
  2227	            merged_coords: List[Any] = []
  2228	            merged_props: List[dict] = []
  2229	            for name, segs in groups.items():
  2230	                lines = []
  2231	                for c, _p in segs:
  2232	                    try:
  2233	                        rounded = [[round(p[0], 5), round(p[1], 5)]
  2234	                                   for p in c if isinstance(p, list) and len(p) >= 2]
  2235	                        if len(rounded) >= 2:
  2236	                            lines.append(LineString([(p[1], p[0]) for p in rounded]))
  2237	                    except Exception:
  2238	                        pass
  2239	                if not lines:
  2240	                    continue
  2241	                try:
  2242	                    union = unary_union(lines)
  2243	                    merged = linemerge(union)
  2244	                    geoms = list(merged.geoms) if merged.geom_type == "MultiLineString" else [merged]
  2245	                    for g in geoms:
  2246	                        if g.geom_type != "LineString" or len(g.coords) < 2:
  2247	                            continue
  2248	                        merged_coords.append([[y, x] for x, y in g.coords])
  2249	                        merged_props.append({
  2250	                            "name": name,
  2251	                            "merged": True,
  2252	                            "length_km": round(g.length * 111.0, 2),
  2253	                        })
  2254	                except Exception:
  2255	                    # 合并失败时保留原线段
  2256	                    for c, p in segs:
  2257	                        merged_coords.append(c)
  2258	                        merged_props.append(p)
  2259	            if merged_coords:
  2260	                layer["coordinates"] = merged_coords
  2261	                layer["properties"] = merged_props
  2262	        return map_layers

  2702	    def _process_waterways(self, way_elements: List[dict]) -> List[dict]:
  2703	        """处理水系要素 - 单线河由细到粗渐变，注记沿河道布局
  2704	
  2705	        制图规范：
  2706	        - 单线河符号由细到粗渐变（河源细线→干流粗线），体现上下游关系；
  2707	        - 河源生成"河源"点状符号；支流汇入主河/湖泊处生成"汇入口"符号，

  2709	        - 湖泊水库按蓝色面渲染；主要河流生成沿河道方向的注记（字头顺流向）。
  2710	        """
  2711	        import math
  2712	        tiers = {}          # tier_index -> [segment_coords,...]
  2713	        tier_names = ["河源细流", "上游河道", "中游河道", "下游河道", "干流主河道"]
  2714	        # 规范一-2：水系统一蓝色#4088C8，河流等级越高线条越粗（干流≈0.5mm/2.4px）
  2715	        tier_weights = [1.2, 1.6, 2.0, 2.4, 2.8]
  2716	        tier_colors = ["#1e90ff", "#1e90ff", "#1e90ff", "#1e90ff", "#1e90ff"]
  2717	        minor = []
  2718	        river_lines = []    # {"coords": [...], "name": str}
  2719	        areas = {}
  2720	        labels = []
  2721	        for elem in way_elements:
  2722	            coords = self._extract_coordinates(elem, "polyline")
  2723	            if not coords or not isinstance(coords, list) or len(coords) < 2:
  2724	                continue
  2725	            tags = elem.get("tags", {})
  2726	            if "waterway" in tags:
  2727	                wtype = tags["waterway"]
  2728	            elif tags.get("natural") == "water":
  2729	                wtype = "lake"
  2730	            elif tags.get("landuse") == "reservoir":
  2731	                wtype = "reservoir"
  2732	            else:
  2733	                wtype = "default"
  2734	            name = tags.get("name", "")

  2736	                if coords[0] != coords[-1]:
  2737	                    coords = coords + [coords[0]]

  2925	    def _point_seg_dist(self, p: list, a: list, b: list) -> float:
  2926	        """点到线段距离（经纬度平面近似，局部小范围足够）"""
  2927	        import math
  2928	        px, py = p[1], p[0]
  2929	        ax, ay = a[1], a[0]
  2930	        bx, by = b[1], b[0]
  2931	        dx, dy = bx - ax, by - ay
  2932	        if dx == 0 and dy == 0:
  2933	            return math.hypot(px - ax, py - ay)
  2934	        t = ((px - ax) * dx + (py - ay) * dy) / (dx * dx + dy * dy)
  2935	        t = max(0.0, min(1.0, t))
  2936	        cx, cy = ax + t * dx, ay + t * dy
  2937	        return math.hypot(px - cx, py - cy)

  2939	    def _max_curvature_vertex(self, coords: list, near: list, window: int = 6) -> list:
  2940	        """在汇入点附近寻找河道曲率最大处的顶点
  2941	
  2942	        曲率用相邻两段线段的转角绝对值近似（弧度）；
  2943	        汇入口与水涯线在曲率最大处相交。
  2944	        """
  2945	        import math
  2946	        n = len(coords)
  2947	        if n < 3:
  2948	            return near
  2949	        closest = min(range(n), key=lambda i: (coords[i][0] - near[0]) ** 2 + (coords[i][1] - near[1]) ** 2)
  2950	        lo, hi = max(1, closest - window), min(n - 2, closest + window)
  2951	        best_i, best_ang = closest, -1.0
  2952	        for i in range(lo, hi + 1):
  2953	            p0, p1, p2 = coords[i - 1], coords[i], coords[i + 1]
  2954	            a1 = math.atan2(p1[0] - p0[0], p1[1] - p0[1])
  2955	            a2 = math.atan2(p2[0] - p1[0], p2[1] - p1[1])
  2956	            d = (a2 - a1 + math.pi) % (2 * math.pi) - math.pi
  2957	            if abs(d) > best_ang:
  2958	                best_ang = abs(d)
  2959	                best_i = i
  2960	        return coords[best_i]

  3410	    def _extract_coordinates(self, elem: dict, layer_type: str) -> Optional[Any]:
  3411	        """从OSM元素中提取坐标
  3412	
  3413	        Args:
  3414	            elem: OSM元素
  3415	            layer_type: 目标图层类型
  3416	
  3417	        Returns:
  3418	            坐标数据：
  3419	                - marker: [lat, lng]
  3420	                - polyline: [[lat, lng], ...]
  3421	            无法提取时返回None
  3422	        """
  3423	        # 辅助函数：检查坐标值是否有效（非None且为数字）
  3424	        def _is_valid_coord(val):
  3425	            return val is not None and isinstance(val, (int, float))
  3426	
  3427	        # way类型有geometry字段（坐标点列表）
  3428	        geometry = elem.get("geometry", [])
  3429	        if geometry:
  3430	            # 过滤掉lat/lon为None或非数字的点
  3431	            coords = [
  3432	                [pt["lat"], pt["lon"]]
  3433	                for pt in geometry
  3434	                if "lat" in pt and "lon" in pt
  3435	                and _is_valid_coord(pt["lat"]) and _is_valid_coord(pt["lon"])
  3436	            ]
  3437	            if coords:
  3438	                if layer_type == "marker" and len(coords) == 1:
  3439	                    return coords[0]
  3440	                return coords
  3441	
  3442	        # node类型直接有lat/lon字段，检查值有效性
  3443	        lat = elem.get("lat")
  3444	        lon = elem.get("lon")
  3445	        if _is_valid_coord(lat) and _is_valid_coord(lon):
  3446	            return [lat, lon]
  3447	
  3448	        return None

  3450	    def _get_default_style(self, key: str) -> dict:
  3451	        """获取图层默认样式
  3452	
  3453	        从MAP_STYLES中获取对应类型的样式配置。
  3454	        highway类型有子类型配置，取默认样式。
  3455	
  3456	        Args:
  3457	            key: 要素类型或图层类型
  3458	
  3459	        Returns:
  3460	            样式字典
  3461	        """
  3462	        style_config = MAP_STYLES.get(key)
  3463	
  3464	        if style_config is None:
  3465	            # 未知类型使用通用默认样式
  3466	            return {"color": "#3388ff", "weight": 2, "opacity": 0.8}
  3467	
  3468	        # highway类型的样式是列表结构（含子类型）
  3469	        if isinstance(style_config, list):
  3470	            for config in style_config:
  3471	                default_style = config.get("default")
  3472	                if default_style:
  3473	                    return dict(default_style)
  3474	            # 回退到列表第一个配置
  3475	            if style_config:
  3476	                return dict(style_config[0].get("default", {"color": "#999999", "weight": 2}))
  3477	
  3478	        # 普通类型直接返回样式字典的副本
  3479	        if isinstance(style_config, dict):
  3480	            return dict(style_config)
  3481	
  3482	        return {"color": "#3388ff", "weight": 2, "opacity": 0.8}
