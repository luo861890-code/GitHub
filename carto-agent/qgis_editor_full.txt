  1337	/** 将图层坐标按目标投影转换并生成 GeoJSON */
  1338	function buildProjectedGeoJson(layer: any, targetEpsg: string): any {
  1339	  const fn = targetEpsg === '3857' ? lonLatToWebMercator : webMercatorToLonLat
  1340	  const gjType = geomTypeToGeoJson(layer?.type || '')
  1341	  const features: any[] = []
  1342	  if (Array.isArray(layer?.features) && layer.features.length > 0) {
  1343	    layer.features.forEach((f: any) => {
  1344	      features.push({
  1345	        type: 'Feature',
  1346	        properties: f?.properties || {},
  1347	        geometry: { type: gjType, coordinates: projectCoordsDeep(f?.coordinates, fn) },
  1348	      })
  1349	    })
  1350	  } else if (layer?.coordinates) {

  1365	function downloadJson(obj: any, filename: string) {
  1366	  const blob = new Blob([JSON.stringify(obj, null, 2)], { type: 'application/geo+json' })
  1367	  const url = URL.createObjectURL(blob)

  1375	/** 投影转换：将选中图层坐标从 WGS84 转换到目标投影并导出 GeoJSON */
  1376	async function runProjection() {
  1377	  const layer = selectedLayer.value ? mapStore.layerGroups[selectedLayer.value]?.data : null
  1378	  if (!layer) {
  1379	    showToast('请先在左侧图层面板选中要转换的图层')
  1380	    return
  1381	  }
  1382	  const input = await showInputDialog({ title: '目标坐标系 EPSG 代码（当前 WGS84 经纬度 4326，可转 3857）', defaultValue: '3857' })
  1383	  if (!input) return
  1384	  const epsg = input.trim().replace(/^epsg:/i, '')
  1385	  if (epsg === '4326') {
  1386	    showToast('当前已是 EPSG:4326 经纬度，无需转换')
  1387	    return
  1388	  }
  1389	  if (epsg !== '3857') {
  1390	    showToast('目前支持 4326 ↔ 3857（Web 墨卡托）')
  1391	    return
  1392	  }
  1393	  const geojson = buildProjectedGeoJson(layer, epsg)
  1394	  downloadJson(geojson, `${layer.name || '图层'}-EPSG${epsg}.geojson`)
  1395	  showToast(`已按 EPSG:${epsg} 转换并导出 GeoJSON`)
  1396	}
